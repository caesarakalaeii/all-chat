<objective>
Wire demand gating into the three leadership-only Go listeners (youtube-listener-innertube, discord-listener, youtube-listener) so they stop polling/publishing for sources without demand. Currently these services subscribe to source:demand and log the demanded set but do not actually gate connections or message publishing.

Purpose: Close DEMAND-08 gap -- all non-Twitch listeners only connect/poll when overlays are open.
Output: Three listeners actively gate their behavior based on demand signals from source-manager.
</objective>

```go
func (m *Manager) reconcileDemand() {
    m.subsMu.Lock()
    defer m.subsMu.Unlock()
    demanded := m.demandedSourceIDs
    if demanded == nil { return } // nil = no filtering (backward compat)
    for slug, ch := range m.subscriptions {
        sourceID := ch.SourceID
        if sourceID == "" { continue }
        if _, ok := demanded[sourceID]; !ok {
            // Source lost demand - unsubscribe
            m.wsClient.Unsubscribe(ch.ChatroomID)
            delete(m.subscriptions, slug)
            // ...
        }
    }
}
```

       ```go
       // UpdateDemandedChannels receives the set of channel IDs that currently have overlay demand.
       // nil means no demand filtering (backward compat). Empty map means zero demand.
       func (m *Manager) UpdateDemandedChannels(demanded map[string]bool) {
           m.demandMu.Lock()
           m.demandedChannels = demanded
           m.demandMu.Unlock()

           m.logger.Info("Demanded channels updated",
               zap.Int("demanded_count", len(demanded)))

           // If demand dropped to zero and no overlays connected, trigger immediate sync
           // to stop all pollers (same pattern as handleOverlayDisconnected's fast path)
           if len(demanded) == 0 {
               m.logger.Info("Zero demand, triggering sync to stop pollers")
               go func() {
                   ctx := context.Background()
                   if err := m.syncStreams(ctx); err != nil {
                       m.logger.Error("Failed to sync after demand drop", zap.Error(err))
                   }
               }()
           }
       }
       ```
