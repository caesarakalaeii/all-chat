**handleSubmit:**
```typescript
async function handleSubmit(e: React.FormEvent) {
  e.preventDefault()
  if (!name.trim()) return
  setIsSubmitting(true)
  try {
    const overlay = await overlaysApi.createOverlay({ name: name.trim() })
    toastManager.add({ title: `"${overlay.name}" created`, type: 'success' })
    router.push(`/overlays/${overlay.id}`)
  } catch (err) {
    toastManager.add({ title: 'Failed to create overlay', description: 'Please try again.', type: 'error' })
  } finally {
    setIsSubmitting(false)
  }
}
```
