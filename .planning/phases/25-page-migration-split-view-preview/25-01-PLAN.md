   ```tsx
   <nav className="sticky top-0 z-50 flex h-[60px] items-center px-8 bg-nav-bg backdrop-blur-[20px] border-b border-border">
     <Link href="/dashboard" className="flex items-center gap-2.5 mr-10 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-twitch rounded-sm">
       <LogoRing aria-hidden="true" />
       <span className="text-base font-extrabold tracking-tight text-text">all-chat</span>
     </Link>
     <div className="flex h-full gap-0.5">
       <Link href="/dashboard" className={isActive('/dashboard') ? activeClass : inactiveClass}>Dashboard</Link>
       {user?.is_admin && (
         <Link href="/admin" className={isActive('/admin') ? activeClass : inactiveClass}>Admin</Link>
       )}
       <Link href="/settings" className={isActive('/settings') ? activeClass : inactiveClass}>Settings</Link>
     </div>
   </nav>
   ```
