SplitView pattern (from RESEARCH.md Pattern 3, locked in CONTEXT.md):
```typescript
// Config panel LEFT ~40%, preview panel RIGHT ~60%
// Draggable divider, min 25% left, max 70% left
// Mobile < 768px: flex-col (config top, preview bottom, divider hidden)
// Uses pointer capture on divider — no setState in pointermove handler
// iframe: src=/overlays/${overlayId}/preview, sandbox="allow-scripts allow-same-origin"

'use client'
import { useRef, useState, useCallback } from 'react'

export function SplitView({ overlayId, children }: { overlayId: string; children: React.ReactNode }) {
  const containerRef = useRef<HTMLDivElement>(null)
  const [leftPct, setLeftPct] = useState(40)
  const isDragging = useRef(false)
  const MIN_LEFT = 25
  const MAX_LEFT = 70

  const onPointerDown = useCallback((e: React.PointerEvent) => {
    e.currentTarget.setPointerCapture(e.pointerId)
    isDragging.current = true
  }, [])

  const onPointerMove = useCallback((e: React.PointerEvent) => {
    if (!isDragging.current || !containerRef.current) return
    const rect = containerRef.current.getBoundingClientRect()
    const pct = ((e.clientX - rect.left) / rect.width) * 100
    setLeftPct(Math.min(MAX_LEFT, Math.max(MIN_LEFT, pct)))
  }, [])

  const onPointerUp = useCallback(() => { isDragging.current = false }, [])

  return (
    <div ref={containerRef} className="flex flex-col md:flex-row h-[calc(100vh-60px)] overflow-hidden"
      onPointerMove={onPointerMove} onPointerUp={onPointerUp}>
      {/* Config panel */}
      <div style={{ width: leftPct + '%' }} className="md:flex-shrink-0 overflow-y-auto w-full md:w-auto">
        {children}
      </div>
      {/* Draggable divider — hidden on mobile */}
      <div
        className="hidden md:block w-1 flex-shrink-0 bg-border hover:bg-twitch/50 cursor-col-resize transition-colors select-none"
        onPointerDown={onPointerDown}
        role="separator"
        aria-label="Drag to resize panels"
        aria-orientation="vertical"
      />
      {/* Preview panel */}
      <div className="flex-1 overflow-hidden bg-bg min-h-[300px] md:min-h-0">
        <iframe
          src={`/overlays/${overlayId}/preview`}
          className="w-full h-full border-0"
          title="Overlay live preview"
          sandbox="allow-scripts allow-same-origin"
        />
      </div>
    </div>
  )
}
```

2. **Config panel width:** On mobile (`< md`): `w-full` (full width, stacked). On desktop (`md:`): `flex-shrink-0` with inline style `width: leftPct%`. Use both `className` for mobile and `style` for desktop via conditional:
   ```tsx
   <div
     className="overflow-y-auto w-full md:flex-shrink-0"
     style={{ ['--split-width' as string]: `${leftPct}%` }}
   >
   ```
   Actually use a simpler approach: apply the inline style only on md+. Since CSS doesn't support media queries in inline styles, use a wrapper approach:
   ```tsx
   <div className="overflow-y-auto">
     <div className="md:hidden">{children}</div>
     <div className="hidden md:block" style={{ width: `${leftPct}%` }}>{children}</div>
   </div>
   ```
   Wait — that double-renders children. Use the correct pattern: apply inline style always but override with CSS on mobile using `md:` — NOT possible with inline styles.

   Use the correct approach: wrap in a div that has `flex-shrink-0 overflow-y-auto` with inline width, and on mobile override with a CSS class that resets width:
   ```tsx
   <div
     ref={configRef}
     className="overflow-y-auto flex-shrink-0 w-full md:w-auto"
     style={{ '--config-width': `${leftPct}%` } as React.CSSProperties}
   >
   ```
   Then in a `<style>` tag or via Tailwind arbitrary: `md:[width:var(--config-width)]`. Use arbitrary Tailwind:
   ```tsx
   <div
     className={`overflow-y-auto flex-shrink-0 w-full md:[width:${leftPct}%]`}
   ```

```tsx
function AddSourceForm({ onAdd }: { onAdd: (platform: string, channelId: string) => Promise<void> }) {
  const [platform, setPlatform] = useState('twitch')
  const [channelId, setChannelId] = useState('')
  const [isAdding, setIsAdding] = useState(false)

  async function handleSubmit(e: React.FormEvent) {
    e.preventDefault()
    if (!channelId.trim()) return
    setIsAdding(true)
    await onAdd(platform, channelId.trim())
    setChannelId('')
    setIsAdding(false)
  }

  return (
    <form onSubmit={handleSubmit} className="border border-border rounded-xl p-4 space-y-3">
      <h3 className="text-sm font-semibold text-text">Add Source</h3>
      <div className="flex gap-2">
        <select
          value={platform}
          onChange={e => setPlatform(e.target.value)}
          className="flex-shrink-0 rounded-lg border border-border bg-surface text-text text-sm px-3 py-2 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-twitch"
          aria-label="Platform"
        >
          <option value="twitch">Twitch</option>
          <option value="youtube">YouTube</option>
          <option value="kick">Kick</option>
          <option value="tiktok">TikTok</option>
        </select>
        <Input
          placeholder="Channel ID or username"
          value={channelId}
          onChange={e => setChannelId(e.target.value)}
          className="flex-1"
          aria-label="Channel ID or username"
        />
      </div>
      <Button variant="gradient" type="submit" disabled={isAdding || !channelId.trim()} className="w-full">
        {isAdding ? <Skeleton className="h-4 w-20" /> : 'Add Source'}
      </Button>
    </form>
  )
}
```

```typescript
async function handleRemoveSource(sourceId: string) {
  try {
    await overlaysApi.removeSource(overlayId, sourceId)
    setSources(prev => prev.filter(s => s.id !== sourceId))
    toastManager.add({ title: 'Source removed', type: 'success' })
  } catch {
    toastManager.add({ title: 'Failed to remove source', description: 'Please try again.', type: 'error' })
  }
}
```

```typescript
async function handleAddSource(platform: string, channelId: string) {
```
