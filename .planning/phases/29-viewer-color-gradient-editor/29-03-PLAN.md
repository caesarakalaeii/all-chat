2. **`storage.ts` — add gradient helpers:**

```typescript
export async function getNameGradient(): Promise<string | null> {
  const storage = await getLocalStorage();
  return storage.viewer_name_gradient ?? null;
}

export async function setNameGradient(gradient: string | null): Promise<void> {
  if (gradient === null) {
    await setLocalStorage({ viewer_name_gradient: undefined });
  } else {
    await setLocalStorage({ viewer_name_gradient: gradient });
  }
}
```

3. **`service-worker.ts` — handle SAVE_NAME_GRADIENT message:**

Find the message handler switch/if chain that handles `SAVE_NAME_COLOR`. Add a matching case for `SAVE_NAME_GRADIENT`:

case 'SAVE_NAME_GRADIENT': {
  const gradientMsg = message as { type: 'SAVE_NAME_GRADIENT'; gradient: string | null };
