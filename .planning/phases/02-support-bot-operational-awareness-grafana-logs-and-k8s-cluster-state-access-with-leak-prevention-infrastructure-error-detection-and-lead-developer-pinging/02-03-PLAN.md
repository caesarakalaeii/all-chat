```yaml
  name: support-bot-cluster-reader
  namespace: allchat
rules:
- apiGroups: [""]
  resources: ["pods", "events"]
  verbs: ["get", "list", "watch"]
- apiGroups: ["apps"]
  resources: ["deployments", "replicasets"]
  verbs: ["get", "list"]
- apiGroups: ["metrics.k8s.io"]
  resources: ["pods"]
  verbs: ["get", "list"]
---
apiVersion: rbac.authorization.k8s.io/v1
kind: RoleBinding
metadata:
  name: support-bot-cluster-reader
  namespace: allchat
subjects:
- kind: ServiceAccount
  name: support-bot
  namespace: allchat
roleRef:
  kind: Role
  name: support-bot-cluster-reader
  apiGroup: rbac.authorization.k8s.io
```
