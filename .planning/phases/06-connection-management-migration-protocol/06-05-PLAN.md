   ```go
   // For each failed pod
   for _, failedPodID := range failedPods {
       // Get channels assigned to failed pod
       assignments, err := c.registry.GetAssignmentsForPod(ctx, failedPodID)
       if err != nil {
           c.logger.Error("Failed to get assignments for failed pod", zap.Error(err))
           continue
       }

       // Trigger migration for each channel
       for _, assignment := range assignments {
           newPodID := c.selectTargetPod(ctx, assignment.SourceID) // Use bounded-load algorithm
           
           event := &coordination.MigrationEvent{
               MigrationID: uuid.NewString(),
               ChannelID:   assignment.SourceID,
               Platform:    c.getPlatformForSource(assignment.SourceID),
               FromPod:     failedPodID,
               ToPod:       newPodID,
               Timestamp:   time.Now(),
               Reason:      "pod_failure",
           }

           if err := c.migrationPublisher.PublishMigrationEvent(ctx, event); err != nil {
               c.logger.Error("Failed to publish migration event", zap.Error(err))
               continue
           }

           // Wait for confirmation before updating registry (MIGRATE-03, MIGRATE-04)
           c.waitForMigrationConfirmation(ctx, event.MigrationID, 60*time.Second)

           // Update assignment registry
           c.registry.StoreAssignment(ctx, assignment.SourceID, newPodID)
       }
   }
   ```

2. **After detecting pod scale-up (new pods in Kubernetes API query):**

   ```go
   // Compute load imbalance
   imbalanceRatio := maxLoad / avgLoad
   ```

**New method: waitForMigrationConfirmation (MIGRATE-03):**
```go
func (c *Coordinator) waitForMigrationConfirmation(ctx context.Context, migrationID string, timeout time.Duration) error {
    // Per CONTEXT.md: "Old pod doesn't disconnect: Coordinator intervention - If old pod doesn't send 'disconnected' confirmation within 60s timeout"
    
    ctx, cancel := context.WithTimeout(ctx, timeout)
    defer cancel()

    // Poll Redis Streams migration:log for confirmation
    for {
        select {
        case <-ctx.Done():
            c.logger.Error("Migration confirmation timeout", zap.String("migration_id", migrationID))
            return fmt.Errorf("timeout waiting for confirmation")
        default:
            // Check for confirmation in Redis Streams
            result, err := c.redisClient.XRead(ctx, &redis.XReadArgs{
                Streams: []string{"migration:log", "0"},
                Count:   100,
            }).Result()
            
            // Parse and check for our migration_id with status="connected"
            // If found, return nil
            // If status="failed", return error
            time.Sleep(1 * time.Second)
        }
    }
}
```

**Why 60s timeout:** Per CONTEXT.md: "If old pod doesn't send 'disconnected' confirmation within 60s timeout - Coordinator marks pod as dead, removes from assignments"
