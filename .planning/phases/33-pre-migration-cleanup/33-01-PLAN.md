<objective>
Normalize source ID suffix handling in both kick-listener and twitch-listener so that
assignedSourceIDs maps always contain bare UUIDs (no ':platform' suffix) at BOTH the
startup extraction and the assignment refresh loop.

Purpose: The SDK ChannelManager interface assumes uniform, bare-UUID keys. The current
kick-listener stores ':kick'-suffixed keys in assignedSourceIDs, causing all channels
to appear unassigned at the manager filter step. The twitch-listener refresh loop already
strips suffixes but its startup extraction does not — this plan makes both paths
consistent in both services.

Output: kick-listener and twitch-listener compile, pass tests, and use identical
strip-at-intake patterns in both code paths.
</objective>

```go
package channels_test

import (
    "strings"
    "testing"

    "github.com/stretchr/testify/assert"
)

// TestManager_SourceIDNormalization verifies that the strip-at-intake pattern
// correctly reduces composite coordinator IDs (e.g. "abc123:kick") to bare UUIDs.
// This is the behavior that must hold in cmd/main.go before passing assignedSourceIDs
// to channels.NewManager.
func TestManager_SourceIDNormalization(t *testing.T) {
    coordinatorSourceIDs := []string{
        "d5e6f7a8-0000-0000-0000-000000000001:kick",
        "d5e6f7a8-0000-0000-0000-000000000002:kick",
        "d5e6f7a8-0000-0000-0000-000000000003",  // already bare — no colon
    }

    assignedIDs := make(map[string]bool)
    for _, raw := range coordinatorSourceIDs {
        sourceID := raw
        if colonIdx := strings.LastIndexByte(sourceID, ':'); colonIdx != -1 {
            sourceID = sourceID[:colonIdx]
        }
        assignedIDs[sourceID] = true
    }

    // Bare UUIDs must be present
    assert.True(t, assignedIDs["d5e6f7a8-0000-0000-0000-000000000001"], "bare UUID 1 should be in map")
    assert.True(t, assignedIDs["d5e6f7a8-0000-0000-0000-000000000002"], "bare UUID 2 should be in map")
    assert.True(t, assignedIDs["d5e6f7a8-0000-0000-0000-000000000003"], "already-bare UUID should be in map")

    // Suffixed keys must NOT be present
    assert.False(t, assignedIDs["d5e6f7a8-0000-0000-0000-000000000001:kick"], "suffixed key must not be in map")
    assert.False(t, assignedIDs["d5e6f7a8-0000-0000-0000-000000000002:kick"], "suffixed key must not be in map")
}
```
