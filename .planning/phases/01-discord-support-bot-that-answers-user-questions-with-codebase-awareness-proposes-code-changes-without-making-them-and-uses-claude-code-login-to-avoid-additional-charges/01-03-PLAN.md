    1. Create `services/support-bot/Dockerfile`:
       ```dockerfile
       FROM node:20-alpine

       WORKDIR /app

       # Install claude CLI globally (provides the `claude` binary for subprocess queries)
       RUN npm install -g @anthropic-ai/claude-code

       # Copy package files
       COPY services/support-bot/package*.json ./

       # Install all dependencies (tsx needed at runtime for TypeScript execution)
       RUN npm ci

       # Copy source code
       COPY services/support-bot/src/ ./src/
       COPY services/support-bot/tsconfig.json ./

       # Run as non-root user
       USER node

       # Start the bot
       CMD ["npm", "start"]
       ```
       Note: tsx is needed at runtime because the start script uses `node --import tsx`. The `claude` CLI is installed globally so the execa subprocess in agent.ts can find it. No @anthropic-ai/claude-agent-sdk is used.

    2. Create `deployments/k8s/base/support-bot/` directory.

    3. Create `deployments/k8s/base/support-bot/deployment.yaml`:
       ```yaml
       apiVersion: apps/v1
       kind: Deployment
       metadata:
         name: support-bot
         namespace: allchat
         labels:
           app: support-bot
       spec:
         replicas: 1
         selector:
           matchLabels:
             app: support-bot
         template:
           metadata:
             labels:
               app: support-bot
           spec:
             initContainers:
               - name: clone-all-chat
                 image: alpine/git
                 command: ['sh', '-c', 'git clone --depth 1 https://$GITHUB_TOKEN@github.com/$GITHUB_OWNER/all-chat.git /repos/all-chat']
                 env:
                   - name: GITHUB_TOKEN
                     valueFrom:
                       secretKeyRef:
                         name: allchat-secrets
                         key: SUPPORT_BOT_GITHUB_TOKEN
                   - name: GITHUB_OWNER
                     value: "moersener"
                 volumeMounts:
                   - name: all-chat-repo
                     mountPath: /repos/all-chat
               - name: clone-all-chat-extension
                 image: alpine/git
                 command: ['sh', '-c', 'git clone --depth 1 https://$GITHUB_TOKEN@github.com/$GITHUB_OWNER/all-chat-extension.git /repos/all-chat-extension']
                 env:
                   - name: GITHUB_TOKEN
                     valueFrom:
                       secretKeyRef:
                         name: allchat-secrets
                         key: SUPPORT_BOT_GITHUB_TOKEN
                   - name: GITHUB_OWNER
                     value: "moersener"
                 volumeMounts:
                   - name: all-chat-extension-repo
                     mountPath: /repos/all-chat-extension
             containers:
               - name: support-bot
                 image: support-bot:latest
                 env:
                   - name: DISCORD_BOT_TOKEN
                     valueFrom:
                       secretKeyRef:
                         name: allchat-secrets
                         key: SUPPORT_BOT_DISCORD_TOKEN
                   - name: CLAUDE_CODE_OAUTH_TOKEN
                     valueFrom:
                       secretKeyRef:
                         name: allchat-secrets
                         key: CLAUDE_CODE_OAUTH_TOKEN
                   - name: GITHUB_TOKEN
                     valueFrom:
                       secretKeyRef:
                         name: allchat-secrets
                         key: SUPPORT_BOT_GITHUB_TOKEN
                   - name: GITHUB_OWNER
                     value: "moersener"
                   - name: ALL_CHAT_REPO_PATH
                     value: "/repos/all-chat"
                   - name: ALL_CHAT_EXTENSION_REPO_PATH
                     value: "/repos/all-chat-extension"
                 volumeMounts:
                   - name: all-chat-repo
                     mountPath: /repos/all-chat
                     readOnly: true
                   - name: all-chat-extension-repo
                     mountPath: /repos/all-chat-extension
                     readOnly: true
                 resources:
                   requests:
                     memory: "128Mi"
                     cpu: "50m"
                   limits:
                     memory: "512Mi"
                     cpu: "500m"
             volumes:
               - name: all-chat-repo
                 emptyDir: {}
               - name: all-chat-extension-repo
                 emptyDir: {}
       ```
       IMPORTANT: Init container commands use `$GITHUB_TOKEN` and `$GITHUB_OWNER` (shell variable syntax), NOT `$(GITHUB_TOKEN)` (Kubernetes command substitution syntax which does not work inside sh -c strings).

       ```yaml
         - deployment.yaml
       ```

    5. Verify Docker build: `cd /home/moersener/Hobby/all-chat && docker build -f services/support-bot/Dockerfile -t support-bot:test .`
