```typescript
ws.onmessage = async (event) => {
  const envelope: WebSocketEnvelope = JSON.parse(event.data);

  // Handle deletion events FIRST (before regular messages)
  if (envelope.type === 'chat_message' && envelope.data.event?.type === 'message_deletion') {
    const deletion = envelope.data.event as DeletionEvent;

    // Update state based on deletion type
    setMessages((prev) => {
      switch (deletion.metadata.deletion_type) {
        case 'single':
          // Remove specific message by internal UUID
          const targetId = deletion.metadata.target_uuid;
          if (!targetId) {
            console.warn('Single deletion missing target_uuid');
            return prev;
          }
          return prev.filter((m) => m.id !== targetId);

        case 'batch':
          // Remove all messages from specific user (timeout/ban)
          const targetUserId = deletion.metadata.target_user_id;
          if (!targetUserId) {
            console.warn('Batch deletion missing target_user_id');
            return prev;
          }
          // Filter out all messages from this user
          return prev.filter((m) => m.user.id !== targetUserId);

        case 'clear':
          // Remove all messages (full chat clear)
          return [];

        default:
          console.warn('Unknown deletion type:', deletion.metadata.deletion_type);
          return prev;
      }
    });

    // Log deletion for debugging
    console.debug('[Deletion]', deletion.metadata.deletion_type, {
      target_uuid: deletion.metadata.target_uuid,
      target_user_id: deletion.metadata.target_user_id,
      target_username: deletion.metadata.target_username,
    });

    return; // Don't process as regular message
  }

  // Handle regular messages (existing code)
  if (envelope.type === 'chat_message' && envelope.data && !envelope.data.event) {
    let message = envelope.data;

    // Existing enrichment code (badges, sorting, etc.)
    message = await resolveTwitchBadgeIcons(message);
    message = sortMessageBadges(message);

    setMessages((prev) => {
      const newMessages = [...prev, message];
      return newMessages.slice(-maxMessages);
    });
  }

  // Handle other envelope types (system, error) if they exist...
};
```
