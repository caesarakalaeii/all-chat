## Project Database Conventions

- Migration numbering: Sequential (001, 002, ..., 029, **030 is next**)
- DOWN migrations: Separate file with `_down.sql` suffix
- Foreign keys: Always `ON DELETE RESTRICT` (no CASCADE)
- Timestamps: Use `TIMESTAMP` type with `DEFAULT NOW()`
- UUIDs: Use `gen_random_uuid()` for primary keys
- Indexes: Partial indexes for boolean flags (`WHERE column = TRUE`)
- Comments: Add `COMMENT ON COLUMN` for documentation

**models/share_request.go:**

```go
package models

import (
	"time"
)

// ShareRequest represents a request to share an overlay between two users
type ShareRequest struct {
	ID               string     `json:"id" db:"id"`
	SenderUserID     string     `json:"sender_user_id" db:"sender_user_id"`
	SenderOverlayID  string     `json:"sender_overlay_id" db:"sender_overlay_id"`
	RecipientUserID  string     `json:"recipient_user_id" db:"recipient_user_id"`
	Status           string     `json:"status" db:"status"` // pending, accepted, rejected, expired
	CreatedAt        time.Time  `json:"created_at" db:"created_at"`
	RespondedAt      *time.Time `json:"responded_at,omitempty" db:"responded_at"`
	ExpiresAt        time.Time  `json:"expires_at" db:"expires_at"`
}

// Valid status constants
const (
	StatusPending  = "pending"
	StatusAccepted = "accepted"
	StatusRejected = "rejected"
	StatusExpired  = "expired"
)

// Validate checks if the share request is valid
func (s *ShareRequest) Validate() error {
	if s.SenderUserID == "" {
		return fmt.Errorf("sender_user_id is required")
	}
	if s.SenderOverlayID == "" {
		return fmt.Errorf("sender_overlay_id is required")
	}
	if s.RecipientUserID == "" {
		return fmt.Errorf("recipient_user_id is required")
	}
	if s.SenderUserID == s.RecipientUserID {
		return fmt.Errorf("cannot share with yourself")
	}
	validStatuses := map[string]bool{
		StatusPending: true, StatusAccepted: true,
		StatusRejected: true, StatusExpired: true,
	}
	if !validStatuses[s.Status] {
		return fmt.Errorf("invalid status: %s", s.Status)
	}
	return nil
}
```
