    1. **AG-01 fix (re-subscribe on channel close):** Add `resubscribe` method:
       ```go
       func (s *Subscriber) resubscribe(overlayID string) {
           // Check if stop signalled
           select {
           case <-s.stopChan:
               return
           default:
           }

           s.mu.Lock()
           defer s.mu.Unlock()

           // Increment reconnect metric per D-14
           if s.metrics != nil {
               s.metrics.PubSubReconnectTotal.WithLabelValues("api-gateway", overlayID).Inc()
           }

           // Close old subscription
           if oldPubsub, exists := s.subscriptions[overlayID]; exists {
               oldPubsub.Close()
           }

           mainChannel := fmt.Sprintf("overlay:%s", overlayID)
           updateChannel := fmt.Sprintf("overlay:%s:updates", overlayID)
           pubsub := s.client.Subscribe(context.Background(), mainChannel, updateChannel)

           if _, err := pubsub.Receive(context.Background()); err != nil {
               s.logger.Error("Failed to re-subscribe",
                   zap.String("overlay_id", overlayID), zap.Error(err))
               delete(s.subscriptions, overlayID)
               return
           }

           s.subscriptions[overlayID] = pubsub
           s.logger.Info("Re-subscribed after channel close",
               zap.String("overlay_id", overlayID))

           s.wg.Add(1)
           go s.listen(context.Background(), overlayID, pubsub)
       }
       ```

    2. **Update listen():** In the `ok == false` branch, replace the simple `return` with:
       ```go
       if !ok {
           s.logger.Warn("Subscription channel closed — re-subscribing",
               zap.String("overlay_id", overlayID))
           go s.resubscribe(overlayID)
           return  // Current goroutine exits, wg.Done() fires via defer
       }
       ```

    3. **AG-02 fix (subscription context):** resubscribe uses `context.Background()` since subscriptions outlive individual HTTP requests. Lifecycle is tied to stopChan, not context.
