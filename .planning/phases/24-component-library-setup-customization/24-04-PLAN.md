Build Dialog and Toast components — the two complex components requiring @base-ui/react primitives for accessibility, and wire Toast into the root layout.

    ```typescript
    "use client"

    import { Dialog as DialogPrimitive } from "@base-ui/react/dialog"
    import { X } from "lucide-react"
    import { cva, type VariantProps } from "class-variance-authority"
    import { cn } from "@/lib/utils"

    const dialogContentVariants = cva(
      "fixed left-1/2 top-1/2 z-50 -translate-x-1/2 -translate-y-1/2 w-full rounded-xl border border-border bg-surface p-6 shadow-xl text-text",
      {
        variants: {
          size: {
            sm:      "max-w-sm",
            default: "max-w-md",
            lg:      "max-w-lg",
          },
        },
        defaultVariants: { size: "default" },
      }
    )

    // Re-export primitive parts for flexible composition
    const DialogRoot = DialogPrimitive.Root
    const DialogTrigger = DialogPrimitive.Trigger
    const DialogPortal = DialogPrimitive.Portal
    const DialogClose = DialogPrimitive.Close
    const DialogTitle = ({ className, ...props }) => (
      <DialogPrimitive.Title className={cn("text-lg font-semibold text-text", className)} {...props} />
    )
    const DialogDescription = ({ className, ...props }) => (
      <DialogPrimitive.Description className={cn("mt-2 text-sm text-text-sub", className)} {...props} />
    )

    // Pre-styled backdrop
    const DialogBackdrop = ({ className, ...props }) => (
      <DialogPrimitive.Backdrop
        className={cn("fixed inset-0 bg-black/60 backdrop-blur-[8px] z-40", className)}
        {...props}
      />
    )

    // Convenience component: includes Portal, Backdrop, and styled Popup
    function DialogContent({
      className,
      size,
      children,
      showCloseButton = true,
      ...props
    }: React.HTMLAttributes<HTMLDivElement> &
      VariantProps<typeof dialogContentVariants> & {
        showCloseButton?: boolean
      }) {
      return (
        <DialogPrimitive.Portal>
          <DialogBackdrop />
          <DialogPrimitive.Popup
            data-slot="dialog-content"
            className={cn(dialogContentVariants({ size, className }))}
            {...props}
          >
            {showCloseButton && (
              <DialogPrimitive.Close
                className="absolute top-4 right-4 rounded-md p-1 text-text-dim hover:bg-surface-2 hover:text-text transition-colors"
                aria-label="Close dialog"
              >
                <X className="size-4" />
              </DialogPrimitive.Close>
            )}
            {children}
          </DialogPrimitive.Popup>
        </DialogPrimitive.Portal>
      )
    }

    export {
      DialogRoot,
      DialogTrigger,
      DialogPortal,
      DialogClose,
      DialogTitle,
      DialogDescription,
      DialogBackdrop,
      DialogContent,
      dialogContentVariants,
    }
    // Also export as namespace for ergonomic usage:
    export const Dialog = {
      Root: DialogRoot,
      Trigger: DialogTrigger,
      Portal: DialogPortal,
      Close: DialogClose,
      Title: DialogTitle,
      Description: DialogDescription,
      Backdrop: DialogBackdrop,
      Content: DialogContent,
    }
    ```
