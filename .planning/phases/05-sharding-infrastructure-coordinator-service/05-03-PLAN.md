    **Structure:**
    ```go
    const (
        HeartbeatKey = "shard:heartbeats"  // Sorted Set: {podID: timestamp}
        HeartbeatTimeout = 15 * time.Second  // User constraint from CONTEXT.md
    )

    type HeartbeatMonitor struct {
        client *redis.Client
        logger *zap.Logger
    }
    ```

    **Functions:**

    1. `NewHeartbeatMonitor(client *redis.Client, logger *zap.Logger) *HeartbeatMonitor`

    2. `PublishHeartbeat(ctx context.Context, podID string) error`:
       - Use ZADD to store pod's current timestamp:
         ```go
         return client.ZAdd(ctx, HeartbeatKey, redis.Z{
             Score:  float64(time.Now().Unix()),
             Member: podID,
         }).Err()
         ```
       - Return error if Redis operation fails
       - Log debug message: "Published heartbeat" with pod_id

    3. `GetFailedPods(ctx context.Context) ([]string, error)`:
       - Calculate cutoff: `time.Now().Add(-15 * time.Second).Unix()`
       - Query pods with heartbeat older than cutoff:
         ```go
         failedPods, err := client.ZRangeByScore(ctx, HeartbeatKey, &redis.ZRangeBy{
             Min: "-inf",
             Max: fmt.Sprintf("%d", cutoff),
         }).Result()
         ```
       - Return list of failed pod IDs
       - Log info message if failures detected: "Detected failed pods" with count

    4. `GetHealthyPods(ctx context.Context) ([]string, error)`:
       - Calculate cutoff: `time.Now().Add(-15 * time.Second).Unix()`
       - Query pods with heartbeat NEWER than cutoff:
         ```go
         healthyPods, err := client.ZRangeByScore(ctx, HeartbeatKey, &redis.ZRangeBy{
             Min: fmt.Sprintf("%d", cutoff),
             Max: "+inf",
         }).Result()
         ```
       - Return list of healthy pod IDs

    5. `CleanupStaleHeartbeats(ctx context.Context) error`:
       - Remove heartbeats older than 5 minutes (pods definitely dead):
         ```go
         cutoff := time.Now().Add(-5 * time.Minute).Unix()
         return client.ZRemRangeByScore(ctx, HeartbeatKey, "-inf", fmt.Sprintf("%d", cutoff)).Err()
         ```
       - Called periodically by coordinator (user constraint: cleanup interval at Claude's discretion)

    6. `RemoveOrphanedAssignments(ctx context.Context, registry *AssignmentRegistry, sourceRepo registry.Repository) error`:
       - Query all assignments from Redis: `registry.GetAllAssignments()`
       - Query all active sources from DB: `sourceRepo.GetActiveSources()`
       - Build sourceID set for fast lookup
       - For each assignment:
         * If source_id NOT in active sources → delete assignment:
           ```go
           registry.DeleteAssignment(ctx, assignment.SourceID)
           ```
       - Log info: "Removed orphaned assignments" with count
       - User constraint: "defense in depth" approach per CONTEXT.md

    Add failure detection step BEFORE assignment computation:

    ```go
    func (c *Coordinator) reconcile(ctx context.Context) {
        ticker := time.NewTicker(c.reconcileInterval)
        defer ticker.Stop()

        for {
            select {
            case <-ticker.C:
                // Step 1: Detect failed pods
                failedPods, err := c.heartbeatMonitor.GetFailedPods(ctx)
                if err != nil {
                    c.logger.Error("Failed to detect failed pods", zap.Error(err))
                } else if len(failedPods) > 0 {
                    c.logger.Info("Detected failed pods, triggering reassignment",
                        zap.Strings("failed_pods", failedPods),
                    )
                    // Failed pods will be excluded from active pod list below
                }

                // Step 2: Query active sources
                sources, err := c.sourceRepo.GetActiveSources(ctx)
                if err != nil {
                    c.logger.Error("Failed to query active sources", zap.Error(err))
                    continue
                }

                // Step 3: Query healthy listener pods from Kubernetes API
                // Filter: phase=Running AND ready=true AND pod_id NOT IN failedPods
                healthyPods, err := c.getHealthyListenerPods(ctx, failedPods)
                if err != nil {
                    c.logger.Error("Failed to query listener pods", zap.Error(err))
                    continue
                }

                if len(healthyPods) == 0 {
                    c.logger.Warn("No healthy listener pods available, skipping reconciliation")
                    continue
                }

                // Step 4: Update assigner with healthy pod list
                c.assigner.UpdatePods(healthyPods)

                // Step 5: Compute and store assignments
                for _, source := range sources {
                    podID, err := c.assigner.AssignChannel(source.ID)
                    if err != nil {
                        c.logger.Error("Failed to assign channel",
                            zap.String("source_id", source.ID),
                            zap.Error(err),
                        )
                        continue
                    }

                    _, err = c.registry.StoreAssignment(ctx, source.ID, podID)
                    if err != nil {
                        c.logger.Error("Failed to store assignment",
                            zap.String("source_id", source.ID),
                            zap.String("pod_id", podID),
                            zap.Error(err),
                        )
                    }
                }

                // Step 6: Cleanup stale heartbeats (every cycle)
                if err := c.heartbeatMonitor.CleanupStaleHeartbeats(ctx); err != nil {
                    c.logger.Error("Failed to cleanup stale heartbeats", zap.Error(err))
                }

                // Step 7: Remove orphaned assignments (every cycle)
                if err := c.heartbeatMonitor.RemoveOrphanedAssignments(ctx, c.registry, c.sourceRepo); err != nil {
                    c.logger.Error("Failed to remove orphaned assignments", zap.Error(err))
                }

                c.logger.Debug("Reconciliation cycle complete",
                    zap.Int("sources", len(sources)),
                    zap.Int("healthy_pods", len(healthyPods)),
                    zap.Int("failed_pods", len(failedPods)),
                )

            case <-c.stopCh:
                c.logger.Info("Reconciliation stopped")
                return
            }
        }
    }
    ```
