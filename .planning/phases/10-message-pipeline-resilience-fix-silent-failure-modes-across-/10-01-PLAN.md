    ```go
    package listener

    import (
        "math/rand"
        "time"
    )

    // JitteredBackoff returns a randomized duration using full jitter algorithm.
    // Formula: random_between(0, min(cap, base * 2^attempt))
    // Base: 1s, Cap: 30s.
    func JitteredBackoff(attempt int) time.Duration {
        const base = time.Second
        const cap = 30 * time.Second
        exp := base
        for i := 0; i < attempt; i++ {
            exp *= 2
            if exp > cap {
                exp = cap
                break
            }
        }
        if exp <= 0 {
            return base
        }
        return time.Duration(rand.Int63n(int64(exp)))
    }
    ```

    ```go
    package zombie

    import (
        "sync/atomic"
        "time"
    )

    // Detector tracks message received-vs-published drift.
    // If messages are received from IRC but publishing has stalled for stallWindow,
    // the pod is considered a zombie and the liveness probe should fail.
    type Detector struct {
        messagesReceived  atomic.Int64
        messagesPublished atomic.Int64
        stallWindow       time.Duration
        lastSnapshot      snapshot
        snapshotTakenAt   time.Time
    }

    type snapshot struct {
        received  int64
        published int64
    }

    // NewDetector creates a zombie detector with the given stall window.
    // Recommended default: 5 minutes.
    func NewDetector(stallWindow time.Duration) *Detector {
        return &Detector{
            stallWindow:   stallWindow,
            snapshotTakenAt: time.Now(),
        }
    }

    func (d *Detector) RecordReceived()  { d.messagesReceived.Add(1) }
    func (d *Detector) RecordPublished() { d.messagesPublished.Add(1) }

    // IsZombie returns true if messages are being received but publishing has stalled.
    // Returns false when both counters are zero (offline channel — D-10).
    func (d *Detector) IsZombie() bool {
        now := time.Now()
        currentReceived := d.messagesReceived.Load()
        currentPublished := d.messagesPublished.Load()

        // Rotate snapshot every stallWindow/2
        if now.Sub(d.snapshotTakenAt) >= d.stallWindow/2 {
            d.lastSnapshot = snapshot{
                received:  currentReceived,
                published: currentPublished,
            }
            d.snapshotTakenAt = now
        }

        // Need at least one full snapshot window to evaluate
        if d.snapshotTakenAt.IsZero() || now.Sub(d.snapshotTakenAt) < d.stallWindow/2 {
            return false
        }

        // Both zero = offline (D-10)
        if currentReceived == 0 && d.lastSnapshot.received == 0 {
            return false
        }

        // Messages received but published stalled
        receivedDelta := currentReceived - d.lastSnapshot.received
        publishedDelta := currentPublished - d.lastSnapshot.published

        return receivedDelta > 0 && publishedDelta == 0
    }
    ```

    - `RecordPublished()` called after successful ring buffer accept (enqueue, not XADD)

    Create ADR-0011 documenting the zombie listener detection architectural decision, following the existing ADR format in the repository.

    The ADR must cover:
    - **Context:** Recurring production outages (April 7, March 29) caused by listener pods that appeared alive (health probes passing) but were not delivering messages. The root cause is that liveness probes only checked process health, not message flow health.
    - **Decision:** Add received-vs-published message drift detection to the twitch-listener liveness probe. Two atomic counters track messages received from IRC and messages published to the ring buffer. If received > 0 but published stalls for 5 minutes, liveness fails and Kubernetes restarts the pod.
    - **Rationale for 5-minute stall window:** Balances detection speed (shorter = faster recovery) against false positives (shorter = more risk during transient Redis blips). 5 minutes gives Redis reconnection backoff time to recover (cap is 30s, so ~10 retry cycles).
    - **False-positive avoidance via both-zero check (D-10):** When a streamer is offline, both received and published are 0. No drift detected, no false kill. This is critical because source-manager keeps sources "active" even when streamers are offline.
    - **Snapshot rotation:** Published counter snapshots are taken every stallWindow/2 (2.5 minutes) to detect stalls within one full window without requiring continuous polling.
    - **Alternatives considered:** Heartbeat-based detection (rejected — adds complexity, doesn't solve the specific drift problem), external monitoring only (rejected — too slow for auto-recovery), readiness probe instead of liveness (rejected — readiness removes from service mesh but doesn't trigger restart).
    - **Consequences:** Pods may restart during genuine extended Redis outages (>5 min). This is acceptable — if Redis is down that long, restarting the pod is a reasonable recovery action. Configurable via `ZOMBIE_STALL_WINDOW_MINUTES` env var.

    Update `docs/adr/README.md` to add the new ADR entry:
