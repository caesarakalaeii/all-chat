       ```go
       func (s *Subscriber) resubscribe(overlayID string) {
           for attempt := 0; ; attempt++ {
               select {
               case <-s.stopChan:
                   s.logger.Info("resubscribe cancelled — stop signal received",
                       zap.String("overlay_id", overlayID))
                   return
               default:
               }

               // Attempt to re-subscribe
               sub := s.redisClient.Subscribe(s.ctx, fmt.Sprintf("overlay:%s", overlayID))
               ch := sub.Channel()

               // ... existing subscription validation logic ...

               if err == nil {
                   // Success — launch new listen goroutine
                   s.logger.Info("resubscribed successfully",
                       zap.String("overlay_id", overlayID),
                       zap.Int("attempt", attempt+1))
                   // Check stopChan before wg.Add to prevent WaitGroup leak (Pitfall 3)
                   select {
                   case <-s.stopChan:
                       sub.Close()
                       return
                   default:
                   }
                   s.wg.Add(1)
                   go s.listen(overlayID, sub, ch)
                   return
               }

               s.logger.Warn("resubscribe attempt failed",
                   zap.String("overlay_id", overlayID),
                   zap.Int("attempt", attempt+1),
                   zap.Error(err))

               sleep := listener.JitteredBackoff(attempt)
               select {
               case <-s.stopChan:
                   return
               case <-time.After(sleep):
               }
           }
       }
       ```

    3. Preserve the existing `wg.Add(1)` before `go s.listen()` pattern. Critical: check `stopChan` BEFORE `wg.Add(1)` to prevent the Pitfall 3 race (see RESEARCH.md).
