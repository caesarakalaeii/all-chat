    ```go
    package listener

    import (
        "os"
        "strconv"
        "time"
    )

    // ListenerConfig holds configurable parameters for ListenerBase.
    // All fields have sensible defaults; use DefaultConfig() to start.
    type ListenerConfig struct {
        // HeartbeatInterval controls how often PublishHeartbeat is called.
        // Default: 30s (matches v1.1 production value).
        HeartbeatInterval time.Duration

        // AssignmentRefreshInterval controls how often QueryAssignments is called.
        // Default: 10s.
        AssignmentRefreshInterval time.Duration

        // StartupJitterMax is the upper bound for random startup delay.
        // Set to 0 to disable jitter (recommended in tests).
        // If 0, the LISTENER_STARTUP_JITTER_MAX env var is NOT read — jitter is fully disabled.
        // Default: 30s (env var LISTENER_STARTUP_JITTER_MAX can override at runtime when using DefaultConfig).
        StartupJitterMax time.Duration

        // DisableCoordinatorFiltering skips assignment-based filtering in UpdateAssignedSourceIDs.
        // Preserves the operational rollback mechanism from twitch-listener (SDK-06).
        DisableCoordinatorFiltering bool

        // OnFatalError is called when a background goroutine encounters a permanent failure.
        // sourceName is the goroutine name ("heartbeat", "assignment-refresh", "migration-subscriber").
        // If nil, goroutines retry indefinitely with exponential backoff (backward-compatible behavior).
        OnFatalError func(source string, err error)
    }

    // DefaultConfig returns a ListenerConfig with production-ready defaults.
    // Reads LISTENER_STARTUP_JITTER_MAX from env (integer seconds); 0 or invalid disables jitter.
    func DefaultConfig() ListenerConfig {
        jitterMax := 30 * time.Second
        if raw := os.Getenv("LISTENER_STARTUP_JITTER_MAX"); raw != "" {
            n, err := strconv.Atoi(raw)
            if err != nil || n <= 0 {
                jitterMax = 0
            } else {
                jitterMax = time.Duration(n) * time.Second
            }
        }
        return ListenerConfig{
            HeartbeatInterval:         30 * time.Second,
            AssignmentRefreshInterval: 10 * time.Second,
            StartupJitterMax:          jitterMax,
        }
    }

    // Env returns the value of the environment variable key,
    // or defaultValue if the variable is unset or empty.
    // Drop-in replacement for the getEnvOrDefault pattern used across all listeners.
    func Env(key, defaultValue string) string {
        if v := os.Getenv(key); v != "" {
            return v
        }
        return defaultValue
    }
    ```

    **shared/listener/base.go** — ListenerBase struct managing 3 goroutine loops.

    IMPORTANT: Define a private `coordinatorClient` interface in base.go BEFORE defining ListenerBase. This interface enables mock injection in tests without exposing an unnecessary public interface. `*coordination.CoordinatorClient` satisfies this interface automatically.

    Add to base.go:
    ```go
    // coordinatorClient abstracts the coordinator HTTP calls used by ListenerBase.
    // *coordination.CoordinatorClient satisfies this interface.
    type coordinatorClient interface {
        PublishHeartbeat(ctx context.Context, podID string) error
        QueryAssignments(ctx context.Context, podID string) ([]*coordination.Assignment, error)
        StartJWTRefresh(ctx context.Context)
        StopJWTRefresh()
    }
    ```

    ListenerBase holds: config ListenerConfig, client coordinatorClient (interface, not concrete type), redisClient *redis.Client (for MigrationSubscriber), logger *zap.Logger, podID string, cancel context.CancelFunc, wg sync.WaitGroup.

    Constructor: `func NewListenerBase(config ListenerConfig, client coordinatorClient, redisClient *redis.Client, podID string, logger *zap.Logger) *ListenerBase`

    `Start(ctx context.Context, mgr ChannelManager) error`:
    1. Apply startup jitter if config.StartupJitterMax > 0: `time.Sleep(time.Duration(rand.Int63n(int64(config.StartupJitterMax))))`
    2. Query initial assignments: call `client.QueryAssignments(ctx, podID)` — block until success (already handles retries internally). Extract source IDs as `map[string]bool`. If DisableCoordinatorFiltering, pass empty map (assignments not filtered).
    3. Call `mgr.UpdateAssignedSourceIDs(assignments)`.
    4. Call `mgr.Start(ctx)`.
    5. Create internal cancel context: `internalCtx, cancel := context.WithCancel(ctx)`, store cancel on b.cancel.
    6. Start 3 goroutines via WaitGroup: `startHeartbeatLoop`, `startAssignmentRefreshLoop`, `startMigrationSubscriberLoop` — each accepts internalCtx.
    7. Start JWT refresh: `client.StartJWTRefresh(internalCtx)`.

    `Stop()`: calls `b.cancel()`, then `b.wg.Wait()`, then `b.client.StopJWTRefresh()`.

    Goroutine loops follow the ticker + select pattern from RESEARCH.md Pattern 1. Backoff pattern for errors: start at 1s, double each failure, cap at 30s, reset to 1s on success.

    `startHeartbeatLoop(ctx, podID)`: ticker at config.HeartbeatInterval. On tick: call client.PublishHeartbeat. On error: log, call config.OnFatalError if set; otherwise retry with backoff.

    `startAssignmentRefreshLoop(ctx, mgr)`: ticker at config.AssignmentRefreshInterval. On tick: call client.QueryAssignments. On success: extract IDs map, call mgr.UpdateAssignedSourceIDs(ids) (skip if DisableCoordinatorFiltering). On error: log + retry with backoff, call OnFatalError if set.

    `startMigrationSubscriberLoop(ctx, mgr)`: If b.redisClient == nil, return immediately (nil guard — allows tests to pass nil redis.Client without real Redis). Otherwise: create MigrationSubscriber with `coordination.NewMigrationSubscriber(b.redisClient, mgr.HandleMigrationEvent, b.logger)`. Call `subscriber.Subscribe(ctx)` in a retry loop — on error, log and retry after backoff delay.

    Import: `"context"`, `"math/rand"`, `"sync"`, `"time"`, `"go.uber.org/zap"`, `"github.com/caesar/all-chat/shared/coordination"`, `"github.com/redis/go-redis/v9"`.

    ```go
    package testutil

    import (
        "context"
        "errors"
        "sync/atomic"

        "github.com/caesar/all-chat/shared/coordination"
    )

    // MockCoordinator is a behavioral in-memory mock for the coordinator client.
    // It tracks call counts and supports failure mode simulation.
    // Does NOT use real HTTP or Redis — safe with goleak.VerifyNone.
    type MockCoordinator struct {
        heartbeatCount  atomic.Int64
        assignmentCount atomic.Int64

        // Failure mode flags — set before calling Start() to simulate failure scenarios
        ShouldFailHeartbeat bool
        ShouldFail401       bool
        ShouldFailTimeout   bool

        // Assignments to return from QueryAssignments (default: empty slice)
        Assignments []*coordination.Assignment
    }

    func (m *MockCoordinator) PublishHeartbeat(_ context.Context, _ string) error {
        m.heartbeatCount.Add(1)
        if m.ShouldFail401 {
            return errors.New("401 unauthorized")
        }
        if m.ShouldFailHeartbeat {
            return errors.New("coordinator down")
        }
        return nil
    }

    func (m *MockCoordinator) QueryAssignments(_ context.Context, _ string) ([]*coordination.Assignment, error) {
        m.assignmentCount.Add(1)
        if m.ShouldFail401 {
            return nil, errors.New("401 unauthorized")
        }
        if m.ShouldFailTimeout {
            return nil, errors.New("request timeout")
        }
        if m.Assignments != nil {
            return m.Assignments, nil
        }
        return []*coordination.Assignment{}, nil
    }

    func (m *MockCoordinator) StartJWTRefresh(_ context.Context) {}
    func (m *MockCoordinator) StopJWTRefresh()                    {}

    // HeartbeatCallCount returns the number of times PublishHeartbeat was called.
    func (m *MockCoordinator) HeartbeatCallCount() int64 { return m.heartbeatCount.Load() }

    // AssignmentCallCount returns the number of times QueryAssignments was called.
    func (m *MockCoordinator) AssignmentCallCount() int64 { return m.assignmentCount.Load() }
    ```
