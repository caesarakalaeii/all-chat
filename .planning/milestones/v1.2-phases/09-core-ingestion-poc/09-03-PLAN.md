Create service entry point with initialization, HTTP server, polling loop integration, and graceful shutdown.

1. **Main Function Structure** (cmd/main.go):
   ```go
   func main() {
       // 1. Environment variables
       logLevel := getEnv("LOG_LEVEL", "info")
       redisHost := getEnv("REDIS_HOST", "localhost")
       redisPort := getEnv("REDIS_PORT", "6379")
       httpPort := getEnv("HTTP_PORT", "8080")
       initialContinuation := getEnv("INITIAL_CONTINUATION", "") // Required for PoC
       channelID := getEnv("CHANNEL_ID", "") // Required for message parsing

       // 2. Logger initialization
       logger := logger.NewLogger("youtube-listener-innertube", logLevel)

       // 3. Redis client
       redisClient := redis.NewClient(&redis.Options{
           Addr: fmt.Sprintf("%s:%s", redisHost, redisPort),
       })
       defer redisClient.Close()

       // 4. InnerTube client (hardcoded API key for PoC)
       innertubeClient := innertube.NewClient("AIzaSyAO_FJ2SlqU8Q4STEHLGCilw_Y9_11qcW8", logger)

       // 5. Publisher
       publisher := publisher.NewStreamPublisher(redisClient, logger)

       // 6. Parser
       parser := innertube.NewParser(logger)

       // 7. Poller
       poller := poller.NewPoller(innertubeClient, parser, initialContinuation, logger)

       // 8. HTTP server with health checks
       router := gin.Default()
       healthHandler := handlers.NewHealthHandler(publisher, innertubeClient)
       router.GET("/health/live", healthHandler.LivenessProbe)
       router.GET("/health/ready", healthHandler.ReadinessProbe)

       srv := &http.Server{
           Addr:    ":" + httpPort,
           Handler: router,
       }

       // 9. Start HTTP server in goroutine
       go func() {
           if err := srv.ListenAndServe(); err != nil && err != http.ErrServerClosed {
               logger.Fatal("HTTP server failed", zap.Error(err))
           }
       }()

       // 10. Start poller in goroutine with message publishing callback
       ctx, cancel := context.WithCancel(context.Background())
       defer cancel()

       go func() {
           for {
               select {
               case <-ctx.Done():
                   return
               default:
                   messages, err := poller.Poll(ctx)
                   if err != nil {
                       // Error already logged by poller, continue
                       continue
                   }
                   for _, msg := range messages {
                       if err := publisher.Publish(ctx, msg); err != nil {
                           logger.Error("Failed to publish message", zap.Error(err))
                       }
                   }
               }
           }
       }()

       // 11. Graceful shutdown on SIGTERM/SIGINT
       quit := make(chan os.Signal, 1)
       signal.Notify(quit, syscall.SIGTERM, syscall.SIGINT)
       <-quit

       logger.Info("Shutting down service...")

       // Stop poller first
       poller.Stop()

       // Shutdown HTTP server with 25s timeout (Kubernetes sends SIGKILL at 30s)
       shutdownCtx, shutdownCancel := context.WithTimeout(context.Background(), 25*time.Second)
       defer shutdownCancel()
       if err := srv.Shutdown(shutdownCtx); err != nil {
           logger.Error("HTTP server forced shutdown", zap.Error(err))
       }

       logger.Info("Service stopped gracefully")
   }
   ```
